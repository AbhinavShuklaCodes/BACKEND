MODULES =>breaking larger files to several other files and export import them to connect 
As your backend grows, putting everything into one file becomes messy.

CommonJS traditionally uses: require() =>import, module.exports =>export

math.js file
function add(a, b) {
    return a + b;
}

module.exports = add;

server.js file
const add = require("./math");

console.log(add(2, 3));

ESM — modern JavaScript modules   uses =>export and import

export default function add(a, b) {
    return a + b;
}

import { add } from "./math.js";

console.log(add(2, 3));
